# Reconstruction skeleton written by `ud init --scaffold`. C++17, CMake >= 3.20, MSVC on Windows, GCC/Clang on Linux.
cmake_minimum_required(VERSION 3.20)
project(reconstruction LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(UD_WITH_SDL "Build the SDL3 platform backend (otherwise the null/headless backend)" ON)
option(UD_FETCH_SDL "Download and build SDL3 when it isn't installed" ON)
# SDL 3.4 refuses to configure on Linux when a default backend's dev package is missing. Install them (see
# `ud tools show sdl3`), or use this for CI / headless oracle runs: dummy and offscreen video only, no audio.
option(UD_SDL_MINIMAL "When fetching SDL3: headless build without desktop, GPU or audio backends" OFF)
set(UD_TARGET "game" CACHE STRING "Name of the rebuilt executable")

if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(WARNING "32-bit build: allowed only as an intermediate step (e.g. DLL injection); the release target is 64-bit")
endif()

add_executable(${UD_TARGET} src/main.cpp)
target_include_directories(${UD_TARGET} PRIVATE src)
# one predictable output folder for every generator: build/bin/<target>[.exe] (`ud run` looks there)
set_target_properties(${UD_TARGET} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")

if(MSVC)
  target_compile_options(${UD_TARGET} PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(${UD_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(${UD_TARGET} PRIVATE -Wall -Wextra)
endif()

set(UD_PLATFORM "null")
if(UD_WITH_SDL)
  find_package(SDL3 CONFIG QUIET)
  if(NOT SDL3_FOUND AND UD_FETCH_SDL)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    if(UD_SDL_MINIMAL)
      set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
      foreach(opt SDL_X11 SDL_WAYLAND SDL_KMSDRM SDL_OPENGL SDL_OPENGLES SDL_VULKAN SDL_GPU SDL_ALSA SDL_PULSEAUDIO SDL_PIPEWIRE
                  SDL_JACK SDL_SNDIO SDL_HIDAPI_LIBUSB SDL_IBUS SDL_DBUS SDL_LIBUDEV SDL_FRIBIDI SDL_LIBTHAI)
        set(${opt} OFF CACHE BOOL "" FORCE)
      endforeach()
    endif()
    FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.4.18 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(SDL3)
  endif()
  if(TARGET SDL3::SDL3)
    set(UD_PLATFORM "sdl3")
  else()
    message(WARNING "SDL3 not found and not fetched: using the null platform backend")
  endif()
endif()

if(UD_PLATFORM STREQUAL "sdl3")
  target_sources(${UD_TARGET} PRIVATE src/platform/platform_sdl3.cpp)
  target_link_libraries(${UD_TARGET} PRIVATE SDL3::SDL3)
else()
  target_sources(${UD_TARGET} PRIVATE src/platform/platform_null.cpp)
endif()
message(STATUS "platform backend: ${UD_PLATFORM}")
